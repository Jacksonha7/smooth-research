\section{Motivation}

\begin{frame}{Why Beamer?}
  \begin{itemize}
    \item One theme controls every slide: fonts, colours, spacing
    \item Equations, citations and code look like the paper
    \item Slides are plain text, so an AI assistant can edit them
    \item Version control with Git, diff every change
  \end{itemize}
  \vspace{1em}
  \begin{alertblock}{Key idea}
    The AI writes \texttt{.tex}; the template decides how it looks.
  \end{alertblock}
\end{frame}

\begin{frame}{Equations come for free}
  A standard attention layer, typeset exactly as in the paper:
  \begin{equation*}
    \mathrm{Attention}(Q,K,V) = \mathrm{softmax}\!\left(\frac{QK^\top}{\sqrt{d_k}}\right)V
  \end{equation*}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{In PowerPoint}
        Equation editor, screenshots, manual alignment.
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{exampleblock}{In Beamer}
        Copy the LaTeX from your paper.
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{PowerPoint vs.\ Beamer}
  \centering
  \begin{tabular}{@{}lll@{}}
    \toprule
    & AI-generated PPT & AI + Beamer template \\
    \midrule
    Layout      & decided by the AI      & fixed by the template \\
    Fonts, colours & often inconsistent  & one theme, every slide \\
    Equations   & images or plain text   & native LaTeX \\
    Editing     & click and drag         & edit text, recompile \\
    Review      & look at each slide     & read the diff \\
    \bottomrule
  \end{tabular}
\end{frame}
